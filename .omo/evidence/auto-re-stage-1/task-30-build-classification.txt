Summary:
  Defined BuildFailureKind enum with 13 variants per spec §12.3:
    Syntax, MissingDeclaration, UnknownType, IncompleteType, LayoutMismatch,
    CallingConventionMismatch, Linkage, MissingSymbol, DuplicateSymbol, Abi,
    UnsupportedPlatformApi, GeneratedCodeDefect, BuildEnvironmentDefect

  Defined RepairStrategy enum with 5 variants:
    CreateWorkItems { kind: WorkItemKind, reason: String },
    BlockWorkItem { reason: String },
    RequestLlmAnalysis { capability: String },
    RequestLayoutInvestigation,
    NoAction

  Implemented classify(diagnostic: &BuildDiagnostic) -> BuildFailureKind:
    - Deterministic MSVC code rules:
      C2065 → MissingDeclaration (or UnsupportedPlatformApi with stdlib context)
      C2061 → UnknownType
      C2079 → IncompleteType
      C2027 → IncompleteType
      C2440 → Abi (or LayoutMismatch with layout/size/offset context)
      C2385 → CallingConventionMismatch
      C3861 → MissingDeclaration
      C1010 → GeneratedCodeDefect
      LNK2019 → MissingSymbol
      LNK2005 → DuplicateSymbol
      LNK* → Linkage
      ENV* → BuildEnvironmentDefect
      (other) → Syntax

  Implemented select_repair_strategy(failure_kind, diagnostic) -> RepairStrategy:
    GeneratedCodeDefect → RequestLlmAnalysis { capability: "llm.analysis.failure" }
    BuildEnvironmentDefect → BlockWorkItem
    LayoutMismatch → RequestLayoutInvestigation
    MissingDeclaration → CreateWorkItems { kind: Generation }
    UnknownType → CreateWorkItems { kind: Generation }
    IncompleteType → CreateWorkItems { kind: Generation }
    CallingConventionMismatch → CreateWorkItems { kind: Generation }
    MissingSymbol → CreateWorkItems { kind: Generation }
    DuplicateSymbol → CreateWorkItems { kind: ConflictResolution }
    Linkage → CreateWorkItems { kind: LinkFailure }
    Abi → CreateWorkItems { kind: Generation }
    UnsupportedPlatformApi → CreateWorkItems { kind: Generation }
    Syntax → CreateWorkItems { kind: BuildFailure }
